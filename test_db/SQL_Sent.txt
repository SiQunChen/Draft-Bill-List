sub get_check_remit {
    my $method = shift;
    my $current_year = shift;
    my $cr;
    my $cr_num;
    my $sql;
    if ($method eq 'A' || $method eq 'C')
    {
         $cr='C';
    }

    if ($method eq 'B' || $method eq 'D' || $method eq 'E' || $method eq 'F'  || $method eq 'G')
    {
       $cr='R';
    }
    if ($cr eq 'C')
    {
       $sql = qq[SELECT check_num AS cr_num FROM  payments WHERE check_num LIKE 'C%' and rec_date >= '$current_year-01-01' AND rec_date <= '$current_year-12-31'   order by check_num desc];
    } else {
       $sql = qq[SELECT remit_num AS cr_num FROM payments WHERE remit_num LIKE 'R%' and rec_date >= '$current_year-01-01' AND rec_date <= '$current_year-12-31' order by remit_num desc];
    }
    my $dbh = HR::DBConnect();
    my $sth = $dbh->prepare ($sql);
    $sth->execute;
    my $payments_result = $sth->fetchrow_hashref;
    $cr_num = $payments_result->{cr_num};
    if ($cr eq 'C')
    {
       $cr_num =~ s/^C0*//;
    } else {
       $cr_num =~ s/^R0*//;
    }
    $cr_num++;
    if ( $cr_num >= 1000 )
    {
       $cr_num = $cr.$cr_num;
    } elsif ( $cr_num >=100 && $cr_num < 1000 ) {
       $cr_num = $cr.'0'.$cr_num;
    } elsif ( $cr_num >=10 && $cr_num < 100 ) {
       $cr_num = $cr.'00'.$cr_num;
    } elsif ( $cr_num >=1 && $cr_num < 10 ) {
       $cr_num = $cr.'000'.$cr_num;
    }

return $cr_num;

}
